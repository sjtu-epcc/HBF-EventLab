set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BASE_BUILD "" CACHE PATH "Sealed integrated llama dependency build; required")
set(HBFLAB_EVENTLAB_SOURCE "${ROOT}/../.." CACHE PATH "Current EventLab backend source root")
get_filename_component(HBFLAB_EVENTLAB_SOURCE "${HBFLAB_EVENTLAB_SOURCE}" REALPATH)

set(LLAMA "${ROOT}/third_party/llama.cpp-integrated")

  message(FATAL_ERROR "Set BASE_BUILD to a configured integrated llama build; build it and seal before running experiments")

file(STRINGS "${BASE_BUILD}/CMakeCache.txt" base_home REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")

get_filename_component(expected_home "${ROOT}/adapters/llama_cpp/integrated" REALPATH)
if(NOT base_home STREQUAL expected_home)
  message(FATAL_ERROR "BASE_BUILD was not configured from this frontend's integrated llama entry")

# Resolve native platform libraries rather than hard-coded Windows .lib names.
foreach(lib llama ggml ggml-base)
  find_library(S_LIBRARY_${lib} NAMES ${lib}
    PATHS "${BASE_BUILD}/llama/src" "${BASE_BUILD}/llama/ggml/src" "${BASE_BUILD}/bin"
    PATH_SUFFIXES "" Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
  add_library(s_${lib} INTERFACE)
  target_link_libraries(s_${lib} INTERFACE "${S_LIBRARY_${lib}}")
endforeach()
find_package(CUDAToolkit REQUIRED)
set(HBFLAB_CUPTI_ROOT "" CACHE PATH "Optional standalone CUPTI SDK root")
find_path(S_CUPTI_INCLUDE cupti.h HINTS "${HBFLAB_CUPTI_ROOT}/include"
  "${CUDAToolkit_ROOT}/include" "${CUDAToolkit_ROOT}/extras/CUPTI/include" REQUIRED)
find_library(S_CUPTI_LIBRARY NAMES cupti HINTS "${HBFLAB_CUPTI_ROOT}/lib" "${HBFLAB_CUPTI_ROOT}/lib/x64"
  "${HBFLAB_CUPTI_ROOT}/lib64" "${CUDAToolkit_ROOT}/lib" "${CUDAToolkit_ROOT}/lib/x64"
  "${CUDAToolkit_ROOT}/lib64" "${CUDAToolkit_ROOT}/extras/CUPTI/lib64" REQUIRED)
include(CheckCXXSourceCompiles)
unset(GPU_S_HAS_KERNEL10 CACHE)
unset(GPU_S_HAS_KERNEL9 CACHE)
set(CMAKE_REQUIRED_INCLUDES "${S_CUPTI_INCLUDE};${CUDAToolkit_INCLUDE_DIRS}")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
check_cxx_source_compiles("#include <cupti.h>\nint main(){CUpti_ActivityKernel10 k{}; return int(k.start);}" GPU_S_HAS_KERNEL10)
if(GPU_S_HAS_KERNEL10)
  set(GPU_S_CUPTI_KERNEL_RECORD CUpti_ActivityKernel10)
else()
  check_cxx_source_compiles("#include <cupti.h>\nint main(){CUpti_ActivityKernel9 k{}; return int(k.start);}" GPU_S_HAS_KERNEL9)
  if(NOT GPU_S_HAS_KERNEL9)
    message(FATAL_ERROR "S experiments require CUPTI with Kernel9 or Kernel10 activity records")
  endif()
  set(GPU_S_CUPTI_KERNEL_RECORD CUpti_ActivityKernel9)
endif()
unset(CMAKE_REQUIRED_INCLUDES)
unset(CMAKE_TRY_COMPILE_TARGET_TYPE)
# Build the current shared backend source, never the upstream vendored backend.
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${HBFLAB_EVENTLAB_SOURCE}" eventlab EXCLUDE_FROM_ALL)

function(s_cuda_target target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE "${ROOT}/include" "${LLAMA}/include"
    "${LLAMA}/ggml/include" "${LLAMA}/vendor" "${S_CUPTI_INCLUDE}")
  target_compile_definitions(${target} PRIVATE GPU_S_CUPTI_KERNEL_RECORD=${GPU_S_CUPTI_KERNEL_RECORD})
  target_link_libraries(${target} PRIVATE s_llama s_ggml s_ggml-base CUDA::cudart "${S_CUPTI_LIBRARY}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc)
  endif()
endfunction()
s_cuda_target(gpu_s_native native.cpp)
s_cuda_target(gpu_s_reference native_reference.cpp)
foreach(target gpu_s_joint gpu_s_transfer_probe gpu_s_segmented gpu_s_s8)
  if(target STREQUAL "gpu_s_s8")
    s_cuda_target(${target} native_s8.cpp)
  else()
    s_cuda_target(${target} native.cpp)
  endif()
  target_compile_definitions(${target} PRIVATE GPU_S_JOINT=1)
  target_link_libraries(${target} PRIVATE HBFEventLab::tensor_bridge)
  if(NOT target STREQUAL "gpu_s_joint")
    target_compile_definitions(${target} PRIVATE GPU_S_TRANSFER_TRACE=1)
  endif()
  if(target STREQUAL "gpu_s_segmented" OR target STREQUAL "gpu_s_s8")
    target_compile_definitions(${target} PRIVATE GPU_S_SEGMENTS=1)
  endif()
endforeach()
add_executable(gpu_s_bridge bridge.cpp)
target_include_directories(gpu_s_bridge PRIVATE "${LLAMA}/vendor")
