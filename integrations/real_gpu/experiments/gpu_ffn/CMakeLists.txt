cmake_minimum_required(VERSION 3.24)
project(gpu_ffn LANGUAGES CXX)
option(GPU_FFN_BUILD_GPU "Build the CUDA FFN executor (OFF permits CPU audit tests)" ON)
if(GPU_FFN_BUILD_GPU)
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/cuda_experiment.cmake")
  s_cuda_target(gpu_ffn ../gpu_s/native_s8.cpp)
  target_compile_definitions(gpu_ffn PRIVATE GPU_S_JOINT=1 GPU_S_TRANSFER_TRACE=1 GPU_S_SEGMENTS=1 GPU_FFN_EXPERIMENT=1)
  target_link_libraries(gpu_ffn PRIVATE HBFEventLab::tensor_bridge)
else()
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  set(HBFLAB_EVENTLAB_SOURCE "${ROOT}/../.." CACHE PATH "Current EventLab backend source root")
  set(LLAMA "${ROOT}/third_party/llama.cpp-integrated")
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${HBFLAB_EVENTLAB_SOURCE}" eventlab EXCLUDE_FROM_ALL)
endif()
add_executable(ffn_audit_tests audit_tests.cpp)
target_include_directories(ffn_audit_tests PRIVATE "${LLAMA}/vendor")
target_link_libraries(ffn_audit_tests PRIVATE hbf_eventlab)
if(MSVC)
  target_compile_options(ffn_audit_tests PRIVATE /utf-8 /EHsc)
endif()
enable_testing()
add_test(NAME ffn_online_audit COMMAND ffn_audit_tests)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_test(NAME ffn_python_contracts COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_LIST_DIR}" -p "test_*.py" -v)

add_executable(ffn_bridge_smoke bridge_smoke.cpp)
target_include_directories(ffn_bridge_smoke PRIVATE "${LLAMA}/vendor")
target_link_libraries(ffn_bridge_smoke PRIVATE HBFEventLab::tensor_bridge)
if(MSVC)
  target_compile_options(ffn_bridge_smoke PRIVATE /utf-8 /EHsc)
endif()
add_test(NAME ffn_main_backend_gate COMMAND ffn_bridge_smoke "${CMAKE_CURRENT_LIST_DIR}/H0-512MiB.ini")
