cmake_minimum_required(VERSION 3.20)
project(HBFUpdatedGPU LANGUAGES CXX)
include(CTest)
add_subdirectory(../EventLab backend)
set(ADAPTER "${CMAKE_CURRENT_SOURCE_DIR}/../adapter")
target_sources(hbf_eventlab PRIVATE "${ADAPTER}/src/gpu/page_controller.cpp")
target_include_directories(hbf_eventlab PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../gpu-headers" "${ADAPTER}/third_party/hbf_sim")
add_executable(gpu_incremental_tests "${ADAPTER}/tests/gpu_incremental.cpp")
target_link_libraries(gpu_incremental_tests PRIVATE HBFEventLab::core)
foreach(scenario backpressure recovery partial too_large streaming cross_unit retire_history burst_backpressure burst_recovery burst_partial burst_too_large burst_streaming burst_cross_unit)
  add_test(NAME gpu_incremental_${scenario} COMMAND gpu_incremental_tests ${scenario})
  set_tests_properties(gpu_incremental_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(probe_current probe_preload.cpp)
target_link_libraries(probe_current PRIVATE HBFEventLab::core)

add_executable(gpu_configuration_tests "${ADAPTER}/tests/gpu_configuration.cpp")
target_link_libraries(gpu_configuration_tests PRIVATE HBFEventLab::core)
foreach(scenario mocs supported address output unknown roundtrip timing write_mode fault limits clock experiment)
  add_test(NAME gpu_configuration_${scenario} COMMAND gpu_configuration_tests ${scenario}
    "${CMAKE_CURRENT_SOURCE_DIR}/../config/gpt2-eventlab-device.ini")
  set_tests_properties(gpu_configuration_${scenario} PROPERTIES TIMEOUT 60)
endforeach()
